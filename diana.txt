<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Couple's Activity Planner</title>
    <script src="https://unpkg.com/react@18/umd/react.development.js"></script>
    <script src="https://unpkg.com/react-dom@18/umd/react-dom.development.js"></script>
    <script src="https://unpkg.com/@babel/standalone/babel.min.js"></script>
    <script src="https://cdn.tailwindcss.com"></script>
    <style>
        .icon { display: inline-block; vertical-align: middle; }
    </style>
</head>
<body>
    <div id="root"></div>

    <script type="text/babel">
        const { useState, useEffect } = React;

        // Simple icon components
        const Heart = ({ size = 24, className = "" }) => (
            <svg width={size} height={size} viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" className={className}>
                <path d="M20.84 4.61a5.5 5.5 0 0 0-7.78 0L12 5.67l-1.06-1.06a5.5 5.5 0 0 0-7.78 7.78l1.06 1.06L12 21.23l7.78-7.78 1.06-1.06a5.5 5.5 0 0 0 0-7.78z"></path>
            </svg>
        );

        const Clock = ({ size = 24, className = "" }) => (
            <svg width={size} height={size} viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" className={className}>
                <circle cx="12" cy="12" r="10"></circle>
                <polyline points="12,6 12,12 16,14"></polyline>
            </svg>
        );

        const Star = ({ size = 24, className = "", fill = "none" }) => (
            <svg width={size} height={size} viewBox="0 0 24 24" fill={fill} stroke="currentColor" strokeWidth="2" className={className}>
                <polygon points="12,2 15.09,8.26 22,9.27 17,14.14 18.18,21.02 12,17.77 5.82,21.02 7,14.14 2,9.27 8.91,8.26"></polygon>
            </svg>
        );

        const Shuffle = ({ size = 24, className = "" }) => (
            <svg width={size} height={size} viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" className={className}>
                <polyline points="16,3 21,3 21,8"></polyline>
                <line x1="4" y1="20" x2="21" y2="3"></line>
                <polyline points="21,16 21,21 16,21"></polyline>
                <line x1="15" y1="15" x2="21" y2="21"></line>
                <line x1="4" y1="4" x2="9" y2="9"></line>
            </svg>
        );

        const Plus = ({ size = 24, className = "" }) => (
            <svg width={size} height={size} viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" className={className}>
                <line x1="12" y1="5" x2="12" y2="19"></line>
                <line x1="5" y1="12" x2="19" y2="12"></line>
            </svg>
        );

        const Filter = ({ size = 24, className = "" }) => (
            <svg width={size} height={size} viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" className={className}>
                <polygon points="22,3 2,3 10,12.46 10,19 14,21 14,12.46"></polygon>
            </svg>
        );

        const CouplesActivityApp = () => {
            const [activities, setActivities] = useState([]);
            const [completedActivities, setCompletedActivities] = useState([]);
            const [filterCategory, setFilterCategory] = useState('all');
            const [showCompleted, setShowCompleted] = useState(false);
            const [customActivity, setCustomActivity] = useState('');
            const [selectedDuration, setSelectedDuration] = useState('any');
            const [suggestedActivity, setSuggestedActivity] = useState(null);

            const activityDatabase = [
                { id: 1, name: "Cook a new recipe together", category: "home", duration: "1-2 hours", difficulty: "easy", description: "Try a cuisine you've never made before" },
                { id: 2, name: "Have a movie marathon", category: "home", duration: "3+ hours", difficulty: "easy", description: "Pick a theme or series to binge together" },
                { id: 3, name: "Create a photo album", category: "home", duration: "1-2 hours", difficulty: "easy", description: "Organize your favorite memories together" },
                { id: 4, name: "Learn a new skill online", category: "home", duration: "2-3 hours", difficulty: "medium", description: "Take an online course or tutorial together" },
                { id: 5, name: "Have an indoor picnic", category: "home", duration: "1 hour", difficulty: "easy", description: "Set up blankets and enjoy a meal on the floor" },
                { id: 6, name: "Go on a hiking trail", category: "outdoor", duration: "2-4 hours", difficulty: "medium", description: "Explore nature and get some exercise" },
                { id: 7, name: "Have a picnic in the park", category: "outdoor", duration: "1-2 hours", difficulty: "easy", description: "Pack lunch and enjoy the outdoors" },
                { id: 8, name: "Visit a farmers market", category: "outdoor", duration: "1-2 hours", difficulty: "easy", description: "Browse local produce and crafts" },
                { id: 9, name: "Go stargazing", category: "outdoor", duration: "2-3 hours", difficulty: "easy", description: "Find a dark spot and observe the night sky" },
                { id: 10, name: "Take a bike ride", category: "outdoor", duration: "1-3 hours", difficulty: "medium", description: "Explore your city or countryside by bike" },
                { id: 11, name: "Paint or draw together", category: "creative", duration: "1-2 hours", difficulty: "easy", description: "Create art side by side or paint each other" },
                { id: 12, name: "Write letters to future selves", category: "creative", duration: "30-60 min", difficulty: "easy", description: "Write letters to read in a year" },
                { id: 13, name: "Create a playlist together", category: "creative", duration: "1 hour", difficulty: "easy", description: "Curate music that represents your relationship" },
                { id: 14, name: "Try pottery or crafts", category: "creative", duration: "2-3 hours", difficulty: "medium", description: "Make something beautiful with your hands" },
                { id: 15, name: "Start a journal together", category: "creative", duration: "30 min", difficulty: "easy", description: "Document your adventures and thoughts" },
                { id: 16, name: "Try a new restaurant", category: "adventure", duration: "1-2 hours", difficulty: "easy", description: "Explore a cuisine you've never tried" },
                { id: 17, name: "Take a day trip", category: "adventure", duration: "4+ hours", difficulty: "medium", description: "Visit a nearby town or attraction" },
                { id: 18, name: "Go to a museum or gallery", category: "adventure", duration: "2-3 hours", difficulty: "easy", description: "Learn something new together" },
                { id: 19, name: "Attend a local event", category: "adventure", duration: "2-4 hours", difficulty: "easy", description: "Check local listings for festivals or shows" },
                { id: 20, name: "Try an escape room", category: "adventure", duration: "1-2 hours", difficulty: "medium", description: "Work together to solve puzzles" },
                { id: 21, name: "Give each other massages", category: "relaxing", duration: "30-60 min", difficulty: "easy", description: "Help each other unwind and relax" },
                { id: 22, name: "Take a bubble bath together", category: "relaxing", duration: "30-60 min", difficulty: "easy", description: "Relax in warm water with candles" },
                { id: 23, name: "Have a tea or coffee tasting", category: "relaxing", duration: "1 hour", difficulty: "easy", description: "Try different varieties and discuss flavors" },
                { id: 24, name: "Do yoga or meditation", category: "relaxing", duration: "30-60 min", difficulty: "easy", description: "Practice mindfulness together" },
                { id: 25, name: "Read books in the same room", category: "relaxing", duration: "1+ hours", difficulty: "easy", description: "Enjoy quiet time while staying connected" }
            ];

            const categories = [
                { id: 'all', name: 'All Activities', color: 'bg-purple-500' },
                { id: 'home', name: 'At Home', color: 'bg-blue-500' },
                { id: 'outdoor', name: 'Outdoor', color: 'bg-green-500' },
                { id: 'creative', name: 'Creative', color: 'bg-pink-500' },
                { id: 'adventure', name: 'Adventure', color: 'bg-orange-500' },
                { id: 'relaxing', name: 'Relaxing', color: 'bg-indigo-500' }
            ];

            useEffect(() => {
                setActivities(activityDatabase);
            }, []);

            const filteredActivities = activities.filter(activity => {
                const categoryMatch = filterCategory === 'all' || activity.category === filterCategory;
                const durationMatch = selectedDuration === 'any' || activity.duration.includes(selectedDuration);
                const completedMatch = showCompleted ? completedActivities.includes(activity.id) : !completedActivities.includes(activity.id);
                return categoryMatch && durationMatch && completedMatch;
            });

            const markComplete = (activityId) => {
                if (completedActivities.includes(activityId)) {
                    setCompletedActivities(completedActivities.filter(id => id !== activityId));
                } else {
                    setCompletedActivities([...completedActivities, activityId]);
                }
            };

            const addCustomActivity = () => {
                if (customActivity.trim()) {
                    const newActivity = {
                        id: Date.now(),
                        name: customActivity,
                        category: 'custom',
                        duration: 'varies',
                        difficulty: 'easy',
                        description: 'Your custom activity'
                    };
                    setActivities([...activities, newActivity]);
                    setCustomActivity('');
                }
            };

            const getRandomActivity = () => {
                const available = filteredActivities.filter(activity => !completedActivities.includes(activity.id));
                if (available.length > 0) {
                    const random = available[Math.floor(Math.random() * available.length)];
                    return random;
                }
                return null;
            };

            const handleRandomSuggestion = () => {
                const random = getRandomActivity();
                setSuggestedActivity(random);
            };

            const getDifficultyColor = (difficulty) => {
                switch (difficulty) {
                    case 'easy': return 'text-green-600 bg-green-100';
                    case 'medium': return 'text-yellow-600 bg-yellow-100';
                    case 'hard': return 'text-red-600 bg-red-100';
                    default: return 'text-gray-600 bg-gray-100';
                }
            };

            const getCategoryColor = (category) => {
                const categoryInfo = categories.find(cat => cat.id === category);
                return categoryInfo?.color || 'bg-gray-500';
            };

            return (
                <div className="min-h-screen bg-gradient-to-br from-pink-50 to-purple-50 p-4">
                    <div className="max-w-4xl mx-auto">
                        {/* Header */}
                        <div className="text-center mb-8">
                            <div className="flex items-center justify-center gap-2 mb-2">
                                <Heart className="text-pink-500" size={32} />
                                <h1 className="text-3xl font-bold text-gray-800">Our Activity Planner</h1>
                                <Heart className="text-pink-500" size={32} />
                            </div>
                            <p className="text-gray-600">Discover fun activities to do together</p>
                        </div>

                        {/* Stats */}
                        <div className="grid grid-cols-1 md:grid-cols-3 gap-4 mb-6">
                            <div className="bg-white rounded-lg p-4 shadow-md text-center">
                                <div className="text-2xl font-bold text-purple-600">{activities.length}</div>
                                <div className="text-gray-600">Total Activities</div>
                            </div>
                            <div className="bg-white rounded-lg p-4 shadow-md text-center">
                                <div className="text-2xl font-bold text-green-600">{completedActivities.length}</div>
                                <div className="text-gray-600">Completed</div>
                            </div>
                            <div className="bg-white rounded-lg p-4 shadow-md text-center">
                                <div className="text-2xl font-bold text-blue-600">{activities.length - completedActivities.length}</div>
                                <div className="text-gray-600">To Try</div>
                            </div>
                        </div>

                        {/* Random Suggestion */}
                        <div className="bg-white rounded-lg p-6 shadow-md mb-6">
                            <div className="flex items-center justify-between mb-4">
                                <h2 className="text-xl font-bold text-gray-800 flex items-center gap-2">
                                    <Shuffle size={20} />
                                    Random Suggestion
                                </h2>
                                <button
                                    onClick={handleRandomSuggestion}
                                    className="bg-purple-500 text-white px-4 py-2 rounded-lg hover:bg-purple-600 transition-colors flex items-center gap-2"
                                >
                                    <Shuffle size={16} />
                                    Get Idea
                                </button>
                            </div>
                            
                            {suggestedActivity ? (
                                <div className="bg-gradient-to-r from-purple-100 to-pink-100 rounded-lg p-4">
                                    <h3 className="font-bold text-lg text-gray-800 mb-2">{suggestedActivity.name}</h3>
                                    <p className="text-gray-600 mb-3">{suggestedActivity.description}</p>
                                    <div className="flex flex-wrap gap-2">
                                        <span className={`px-2 py-1 rounded text-sm ${getCategoryColor(suggestedActivity.category)} text-white`}>
                                            {categories.find(cat => cat.id === suggestedActivity.category)?.name || suggestedActivity.category}
                                        </span>
                                        <span className="px-2 py-1 rounded text-sm bg-gray-100 text-gray-700 flex items-center gap-1">
                                            <Clock size={12} />
                                            {suggestedActivity.duration}
                                        </span>
                                        <span className={`px-2 py-1 rounded text-sm ${getDifficultyColor(suggestedActivity.difficulty)}`}>
                                            {suggestedActivity.difficulty}
                                        </span>
                                    </div>
                                </div>
                            ) : (
                                <p className="text-gray-500 text-center py-4">Click "Get Idea" for a random activity suggestion!</p>
                            )}
                        </div>

                        {/* Add Custom Activity */}
                        <div className="bg-white rounded-lg p-6 shadow-md mb-6">
                            <h2 className="text-xl font-bold text-gray-800 mb-4 flex items-center gap-2">
                                <Plus size={20} />
                                Add Your Own Activity
                            </h2>
                            <div className="flex gap-2">
                                <input
                                    type="text"
                                    value={customActivity}
                                    onChange={(e) => setCustomActivity(e.target.value)}
                                    placeholder="Enter a custom activity..."
                                    className="flex-1 px-4 py-2 border border-gray-300 rounded-lg focus:outline-none focus:ring-2 focus:ring-purple-500"
                                    onKeyPress={(e) => e.key === 'Enter' && addCustomActivity()}
                                />
                                <button
                                    onClick={addCustomActivity}
                                    className="bg-purple-500 text-white px-4 py-2 rounded-lg hover:bg-purple-600 transition-colors"
                                >
                                    Add
                                </button>
                            </div>
                        </div>

                        {/* Filters */}
                        <div className="bg-white rounded-lg p-6 shadow-md mb-6">
                            <h2 className="text-xl font-bold text-gray-800 mb-4 flex items-center gap-2">
                                <Filter size={20} />
                                Filters
                            </h2>
                            
                            <div className="space-y-4">
                                {/* Category Filter */}
                                <div>
                                    <label className="block text-sm font-medium text-gray-700 mb-2">Category</label>
                                    <div className="flex flex-wrap gap-2">
                                        {categories.map(category => (
                                            <button
                                                key={category.id}
                                                onClick={() => setFilterCategory(category.id)}
                                                className={`px-3 py-2 rounded-lg text-sm transition-colors ${
                                                    filterCategory === category.id
                                                        ? `${category.color} text-white`
                                                        : 'bg-gray-100 text-gray-700 hover:bg-gray-200'
                                                }`}
                                            >
                                                {category.name}
                                            </button>
                                        ))}
                                    </div>
                                </div>

                                {/* Duration Filter */}
                                <div>
                                    <label className="block text-sm font-medium text-gray-700 mb-2">Duration</label>
                                    <select
                                        value={selectedDuration}
                                        onChange={(e) => setSelectedDuration(e.target.value)}
                                        className="px-4 py-2 border border-gray-300 rounded-lg focus:outline-none focus:ring-2 focus:ring-purple-500"
                                    >
                                        <option value="any">Any Duration</option>
                                        <option value="30">30 minutes or less</option>
                                        <option value="1">1-2 hours</option>
                                        <option value="2">2-3 hours</option>
                                        <option value="3">3+ hours</option>
                                    </select>
                                </div>

                                {/* Show Completed */}
                                <div className="flex items-center gap-2">
                                    <input
                                        type="checkbox"
                                        id="showCompleted"
                                        checked={showCompleted}
                                        onChange={(e) => setShowCompleted(e.target.checked)}
                                        className="rounded border-gray-300 text-purple-600 focus:ring-purple-500"
                                    />
                                    <label htmlFor="showCompleted" className="text-sm font-medium text-gray-700">
                                        Show completed activities
                                    </label>
                                </div>
                            </div>
                        </div>

                        {/* Activities Grid */}
                        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-6">
                            {filteredActivities.map(activity => (
                                <div key={activity.id} className="bg-white rounded-lg shadow-md overflow-hidden hover:shadow-lg transition-shadow">
                                    <div className="p-6">
                                        <div className="flex items-start justify-between mb-3">
                                            <h3 className="font-bold text-lg text-gray-800 flex-1">{activity.name}</h3>
                                            <button
                                                onClick={() => markComplete(activity.id)}
                                                className={`ml-2 p-1 rounded transition-colors ${
                                                    completedActivities.includes(activity.id)
                                                        ? 'text-yellow-500 hover:text-yellow-600'
                                                        : 'text-gray-300 hover:text-yellow-500'
                                                }`}
                                            >
                                                <Star size={20} fill={completedActivities.includes(activity.id) ? 'currentColor' : 'none'} />
                                            </button>
                                        </div>
                                        
                                        <p className="text-gray-600 mb-4">{activity.description}</p>
                                        
                                        <div className="flex flex-wrap gap-2">
                                            <span className={`px-2 py-1 rounded text-sm ${getCategoryColor(activity.category)} text-white`}>
                                                {categories.find(cat => cat.id === activity.category)?.name || activity.category}
                                            </span>
                                            <span className="px-2 py-1 rounded text-sm bg-gray-100 text-gray-700 flex items-center gap-1">
                                                <Clock size={12} />
                                                {activity.duration}
                                            </span>
                                            <span className={`px-2 py-1 rounded text-sm ${getDifficultyColor(activity.difficulty)}`}>
                                                {activity.difficulty}
                                            </span>
                                        </div>
                                    </div>
                                </div>
                            ))}
                        </div>

                        {filteredActivities.length === 0 && (
                            <div className="text-center py-12">
                                <p className="text-gray-500 text-lg">No activities found with current filters.</p>
                                <p className="text-gray-400">Try adjusting your filters or add a custom activity!</p>
                            </div>
                        )}
                    </div>
                </div>
            );
        };

        ReactDOM.render(<CouplesActivityApp />, document.getElementById('root'));
    </script>
</body>
</html>
